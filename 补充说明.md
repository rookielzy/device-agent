## 登录获取鉴权信息

先写死用 账号：18826224869、密码：123abc 作为登录信息获取鉴权信息。后端已有的登录接口如下：

```ts
export interface AccountPasswordLoginReq {
  mobile: string;
  password: string;
}

export interface AccountPasswordLoginResp {
  access_token: string;
  token_type: string;
  refresh_token: string;
  expires_in: number;
  scope: string;
  account_expires: number;
}

export function loginWithPassword(
  data: AccountPasswordLoginReq,
  options?: UserCenterRequestOptions,
) {
  return http.post<AccountPasswordLoginResp, AccountPasswordLoginReq>({
    url: buildUserCenterUrl("/oauth/login"),
    data,
    auth: "none",
    ...buildUserCenterRequestOptions(options),
  });
}
```
